cmake_minimum_required(VERSION 3.20)
project(lithrez VERSION 2.0 LANGUAGES CXX)

# lithrez (command line) and LithRez Studio (Qt 6), on a shared library of
# LithTech formats. Builds with clang, gcc or MSVC on Linux, macOS and
# Windows.

# The version the binaries report: a release tag's, or "dev".
set(LITHTOOLS_VERSION "dev" CACHE STRING "Version string the tools report")

option(LITHREZ_BUILD_GUI "Build LithRez Studio (needs Qt 6)" ON)
option(LITHREZ_BUILD_TESTS "Build the tests" ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
  add_compile_options(-Wall -Wextra -Wno-unused-parameter)
endif()

add_library(lithtech STATIC
  core/src/rez.cpp
  core/src/dtx.cpp
  core/src/abc.cpp
  core/src/ltb.cpp
  core/src/ps2ltb.cpp
  ${CMAKE_CURRENT_BINARY_DIR}/ps2ltb_names.cpp
  core/src/pe_strings.cpp
  core/src/sound.cpp
)
target_include_directories(lithtech PUBLIC core/include)
# The names for PS2 models' hashes: the Blender extension's list, compiled in.
set(PS2LTB_NAMES ${CMAKE_CURRENT_SOURCE_DIR}/../../plugins/blender/io_scene_lithtech/lithtech/ps2ltb_names.txt)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PS2LTB_NAMES})
file(STRINGS ${PS2LTB_NAMES} _ps2_names)
set(_ps2_out "// Generated by CMake from ps2ltb_names.txt.\n#include <cstddef>\nnamespace lithtech::ps2ltb {\nextern const char* const kNames[] = {\n")
set(_ps2_n 0)
foreach(_name IN LISTS _ps2_names)
  if(_name STREQUAL "" OR _name MATCHES "^#")
    continue()
  endif()
  string(REPLACE "\\" "\\\\" _name "${_name}")
  string(REPLACE "\"" "\\\"" _name "${_name}")
  string(APPEND _ps2_out "    \"${_name}\",\n")
  math(EXPR _ps2_n "${_ps2_n} + 1")
endforeach()
string(APPEND _ps2_out "};\nextern const std::size_t kNameCount = ${_ps2_n};\n}\n")
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/ps2ltb_names.cpp CONTENT "${_ps2_out}" @ONLY)
# minimp3 (CC0), for the MP3 sounds; see third_party/minimp3/README.md.
target_include_directories(lithtech SYSTEM PRIVATE third_party/minimp3)

add_executable(lithrez cli/lithrez.cpp)
target_link_libraries(lithrez PRIVATE lithtech)
target_compile_definitions(lithrez PRIVATE LITHREZ_VERSION="${LITHTOOLS_VERSION}")
install(TARGETS lithrez RUNTIME DESTINATION bin)

if(LITHREZ_BUILD_GUI)
  find_package(Qt6 6.2 COMPONENTS Widgets OpenGL OpenGLWidgets Multimedia)
  if(Qt6_FOUND)
    add_subdirectory(gui)
  else()
    message(WARNING "Qt 6 (with Widgets, OpenGL and Multimedia) not found: LithRez Studio will not be built")
  endif()
endif()

if(LITHREZ_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
